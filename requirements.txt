# --- Runtime dependencies ---
fastapi                  # web framework (auth, upload, /chat API)
uvicorn[standard]        # ASGI server that runs the FastAPI app
python-multipart         # lets FastAPI accept file uploads
pydantic                 # request/response validation (comes with FastAPI)
fastembed                # ONNX embedding model (all-MiniLM-L6-v2), low memory
faiss-cpu                # vector database for similarity search
pdfplumber               # PDF text extraction
groq                     # official Groq Python client (LLM generation)
pymongo                  # MongoDB driver (users, document metadata, preferences)
bcrypt                   # password hashing (plaintext never stored)
pyjwt                    # JWT session tokens for login
python-dotenv            # loads secrets from .env
numpy                    # array math used by FAISS + embeddings
